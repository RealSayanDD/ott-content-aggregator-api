
from sqlalchemy.orm import Session
from sqlalchemy import or_
from . import models, schemas


def create_content(db: Session, data: schemas.ContentCreate):
    content = models.Content(**data.model_dump())
    db.add(content)
    db.commit()
    db.refresh(content)
    return content


def get_contents(
    db: Session,
    title: str = None,
    genre: str = None,
    content_type: str = None,
    release_year: int = None,
    platform: str = None,
):
    query = db.query(models.Content)

    if title:
        query = query.filter(
            models.Content.title.ilike(f"%{title}%")
        )

    if genre:
        query = query.filter(
            models.Content.genre.ilike(f"%{genre}%")
        )

    if content_type:
        query = query.filter(
            models.Content.content_type == content_type
        )

    if release_year:
        query = query.filter(
            models.Content.release_year == release_year
        )

    if platform:
        query = query.filter(
            models.Content.platform.ilike(f"%{platform}%")
        )

    return query.order_by(models.Content.title).all()


def get_content(db: Session, content_id: int):
    return db.query(models.Content).filter(
        models.Content.id == content_id
    ).first()


def update_content(
    db: Session,
    content: models.Content,
    data: schemas.ContentCreate
):
    for key, value in data.model_dump().items():
        setattr(content, key, value)

    db.commit()
    db.refresh(content)
    return content


def delete_content(db: Session, content: models.Content):
    db.delete(content)
    db.commit()
